PHP PDO Database & JSON API (Task 21)

Click button to request API

PHP Code:

<?php
header('Content-Type: application/json');

// Connect to MySQL using PDO
$pdo = new PDO("mysql:host=localhost;dbname=school;charset=utf8", "root", "");

// Prepared query
$stmt = $pdo->prepare("SELECT id, name, email FROM students LIMIT 5");
$stmt->execute();
$data = $stmt->fetchAll(PDO::FETCH_ASSOC);

echo json_encode([
    "status" => 200,
    "message" => "Data fetched successfully",
    "students" => $data
]);
?>
